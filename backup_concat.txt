class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Granizo Demo',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        colorScheme: ColorScheme.fromSeed(seedColor: Colors.blue),
        useMaterial3: true,
      ),
      home: const MapPage(),
    );
  }
}

class MapPage extends StatefulWidget {
  const MapPage({super.key});

  @override
  State<MapPage> createState() => _MapPageState();
}

class _MapPageState extends State<MapPage> {
  final MapController _mapController = MapController();
  final LatLng _start = const LatLng(-34.6037, -58.3816); // Obelisco BA aprox.

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Stack(
        children: [
          // IMPORTANTE: sin 'const' acÃ¡ ni en la lista de children.
          FlutterMap(
            mapController: _mapController,
            options: const MapOptions(
              initialCenter: LatLng(-34.6037, -58.3816),
              initialZoom: 13,
            ),
            children: [
              TileLayer(
                urlTemplate:
                'https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png',
                subdomains: const ['a', 'b', 'c'],
                userAgentPackageName: 'com.example.granizo_final',
              ),
            ],
          ),

          // Tres botones flotantes apilados
          Positioned(
            right: 16,
            bottom: 16,
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                FloatingActionButton.extended(
                  heroTag: 'btn1',
                  onPressed: () {
                    _mapController.move(_start, 13);
                  },
                  icon: const Icon(Icons.my_location),
                  label: const Text('Centrar'),
                ),
                const SizedBox(height: 12),
                FloatingActionButton.extended(
                  heroTag: 'btn2',
                  onPressed: () {
                    _mapController.move(_mapController.camera.center,
                        _mapController.camera.zoom + 1);
                  },
                  icon: const Icon(Icons.zoom_in),
                  label: const Text('Zoom +'),
                ),
                const SizedBox(height: 12),
                FloatingActionButton.extended(
                  heroTag: 'btn3',
                  onPressed: () {
                    _mapController.move(_mapController.camera.center,
                        _mapController.camera.zoom - 1);
                  },
                  icon: const Icon(Icons.zoom_out),
                  label: const Text('Zoom -'),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }
}
